# MRP API - Manual Test Commands
# Copy and paste these commands one by one

## Setup
BASE_URL=http://localhost:8080

## ===================================
## 1. AUTHENTICATION
## ===================================

# Register User 1
curl -X POST http://localhost:8080/api/users/register \
  -H "Content-Type: application/json" \
  -d '{"username":"testuser1","password":"pass123"}'

# Register User 2
curl -X POST http://localhost:8080/api/users/register \
  -H "Content-Type: application/json" \
  -d '{"username":"testuser2","password":"pass456"}'

# Login User 1 (SAVE THE TOKEN!)
curl -X POST http://localhost:8080/api/users/login \
  -H "Content-Type: application/json" \
  -d '{"username":"testuser1","password":"pass123"}'
# Response: {"token":"testuser1-mrpToken"}
# SET THIS: TOKEN1="testuser1-mrpToken"

# Login User 2 (SAVE THE TOKEN!)
curl -X POST http://localhost:8080/api/users/login \
  -H "Content-Type: application/json" \
  -d '{"username":"testuser2","password":"pass456"}'
# Response: {"token":"testuser2-mrpToken"}
# SET THIS: TOKEN2="testuser2-mrpToken"

# Login with Wrong Password (should fail)
curl -X POST http://localhost:8080/api/users/login \
  -H "Content-Type: application/json" \
  -d '{"username":"testuser1","password":"wrongpassword"}'

## ===================================
## 2. MEDIA MANAGEMENT
## ===================================

# Create Media Entry (SAVE THE MEDIA_ID from response!)
curl -X POST http://localhost:8080/api/media \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer testuser1-mrpToken" \
  -d '{
    "title": "Inception",
    "description": "A mind-bending thriller",
    "mediaType": "MOVIE",
    "releaseYear": 2010,
    "genres": ["Sci-Fi", "Thriller"],
    "ageRestriction": 13
  }'
# Response will contain: "id":"SOME-UUID"
# SET THIS: MEDIA_ID="SOME-UUID"

# Get All Media
curl -X GET http://localhost:8080/api/media

# Get Media by ID (replace MEDIA_ID)
curl -X GET http://localhost:8080/api/media/MEDIA_ID

# Search Media by Title
curl -X GET "http://localhost:8080/api/media/search?title=Inception"

# Filter by Genre
curl -X GET "http://localhost:8080/api/media?genre=Sci-Fi"

# Filter by Type
curl -X GET "http://localhost:8080/api/media?type=MOVIE"

# Filter by Year
curl -X GET "http://localhost:8080/api/media?year=2010"

# Filter by Age Restriction
curl -X GET "http://localhost:8080/api/media?age=13"

# Sort by Rating
curl -X GET "http://localhost:8080/api/media?sort=rating"

# Combined Filters
curl -X GET "http://localhost:8080/api/media?genre=Sci-Fi&type=MOVIE&sort=rating"

# Update Media (Owner - replace MEDIA_ID)
curl -X PUT http://localhost:8080/api/media/MEDIA_ID \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer testuser1-mrpToken" \
  -d '{
    "title": "Inception - Updated",
    "description": "Updated description",
    "mediaType": "MOVIE",
    "releaseYear": 2010,
    "genres": ["Sci-Fi", "Action"],
    "ageRestriction": 16
  }'

# Update Media by Non-Owner (should fail - replace MEDIA_ID)
curl -X PUT http://localhost:8080/api/media/MEDIA_ID \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer testuser2-mrpToken" \
  -d '{
    "title": "Hacked Title",
    "description": "Should not work",
    "mediaType": "MOVIE",
    "releaseYear": 2010,
    "genres": ["Action"],
    "ageRestriction": 12
  }'

## ===================================
## 3. RATINGS
## ===================================

# Create Rating (SAVE THE RATING_ID from response!)
curl -X POST http://localhost:8080/api/ratings \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer testuser2-mrpToken" \
  -d '{
    "mediaId": "MEDIA_ID",
    "stars": 5,
    "comment": "Amazing movie! Highly recommend!"
  }'
# Response will contain: "id":"SOME-UUID"
# SET THIS: RATING_ID="SOME-UUID"

# Create Rating with Invalid Stars (should fail)
curl -X POST http://localhost:8080/api/ratings \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer testuser1-mrpToken" \
  -d '{
    "mediaId": "MEDIA_ID",
    "stars": 6,
    "comment": "Invalid rating"
  }'

# Get Ratings for Media (replace MEDIA_ID)
curl -X GET http://localhost:8080/api/ratings/media/MEDIA_ID

# Get Ratings by User (replace USER_ID)
curl -X GET http://localhost:8080/api/ratings/user/USER_ID

# Update Rating (replace RATING_ID)
curl -X PUT http://localhost:8080/api/ratings/RATING_ID \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer testuser2-mrpToken" \
  -d '{
    "stars": 4,
    "comment": "Updated comment - still great but not perfect"
  }'

# Confirm Rating (replace RATING_ID)
curl -X POST http://localhost:8080/api/ratings/RATING_ID/confirm \
  -H "Authorization: Bearer testuser2-mrpToken"

# Like a Rating (replace RATING_ID)
curl -X POST http://localhost:8080/api/ratings/RATING_ID/like \
  -H "Authorization: Bearer testuser1-mrpToken"

# Delete Rating (replace RATING_ID)
curl -X DELETE http://localhost:8080/api/ratings/RATING_ID \
  -H "Authorization: Bearer testuser2-mrpToken"
